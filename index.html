function setupDrag() { const marker = document.getElementById('marker'); const line = document.querySelector('.line-wrapper'); let isDragging = false; const move = (e) => { if(!isDragging) return; let rect = line.getBoundingClientRect(); let clientX = e.touches ? e.touches[0].clientX : e.clientX; let x = clientX - rect.left; let pct = Math.max(0, Math.min(100, (x / rect.width) * 100)); // Arrotondiamo allo scatto del 10% (0, 10, 20... 100) let snapPct = Math.round(pct / 10) * 10; marker.style.left = snapPct + '%'; }; marker.onmousedown = marker.ontouchstart = (e) => { isDragging = true; e.preventDefault(); }; window.onmousemove = window.ontouchmove = move; window.onmouseup = window.ontouchend = () => isDragging = false; } function check() { const f = document.getElementById('feedback'); let ok = false; if(mode === 4) { let m = document.getElementById('marker'); // Recuperiamo la percentuale e calcoliamo il valore con 1 decimale di precisione let currentPct = parseFloat(m.style.left); let currentVal = 3 + (currentPct / 100); // Usiamo Math.round per evitare i decimali infiniti di JS (es 3.900000004) let valArrotondato = Math.round(currentVal * 10) / 10; // Confronto finale ok = (valArrotondato === target); } else { const num = parseInt(document.getElementById('num-in').value); const dec = document.getElementById('dec-in').value.replace(',', '.'); let correctDec = (target / document.getElementById('den-in').value); ok = (num === target && Math.abs(parseFloat(dec) - correctDec) < 0.0001); } if(ok) { f.innerHTML = "🌟 Bravissimo!"; f.style.color = "var(--success)"; document.getElementById('check-btn').classList.add('hidden'); document.getElementById('next-btn').classList.remove('hidden'); } else { f.innerHTML = "❌ Riprova!"; f.style.color = "var(--danger)"; } }
